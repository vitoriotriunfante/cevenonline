let CANAL = 'VAREJO';
try { const c = localStorage.getItem('brasileirao_canal'); if (c === 'AS' || c === 'VAREJO') CANAL = c; } catch (e) {}
const AS_SEMANAS = [{ n: 1, de: 1, ate: 7 }, { n: 2, de: 8, ate: 14 }, { n: 3, de: 15, ate: 20 }, { n: 4, de: 21, ate: 31 }];
function semanaAS(diaISO) {
  const d = parseInt(String(diaISO).slice(8, 10), 10);
  const s = AS_SEMANAS.find(x => d <= x.ate) || AS_SEMANAS[3];
  const mes = String(diaISO).slice(0, 7), ult = new Date(Date.UTC(+mes.slice(0, 4), +mes.slice(5, 7), 0)).getUTCDate();
  const ate = Math.min(s.ate, ult), p = (n) => String(n).padStart(2, '0');
  const dias = [];
  for (let k = s.de; k <= ate; k++) { const iso = mes + '-' + p(k), dt = new Date(iso + 'T12:00:00Z'), w = dt.getUTCDay(); if (w !== 0 && w !== 6) dias.push(iso); }
  return { n: s.n, de: mes + '-' + p(s.de), ate: mes + '-' + p(ate), dias };
}
function rcasDoAS() { return new Set(((DADOS && DADOS.vendedores) || []).filter(v => v.canal === 'AS').map(v => String(v.rca))); }
function lanceDoCanal(l, setAS) { const eAS = setAS.has(String(l.rca)); return CANAL === 'AS' ? eAS : !eAS; }
const ddmm = (iso) => String(iso).slice(8, 10) + '/' + String(iso).slice(5, 7);

// Busca os lances do período do canal: VAREJO = o dia escolhido; AS = a SEMANA (dias úteis até hoje) que contém o dia escolhido. Só lances do canal.
async function buscarLancesPeriodo(dia, filial) {
  const hoje = hojeISOBrasilia();
  const per = CANAL === 'AS' ? semanaAS(dia) : { n: null, de: dia, ate: dia, dias: [dia] };
  const dias = per.dias.filter(d => d <= hoje);
  const setAS = rcasDoAS();
  let lances = [];
  const resp = await Promise.all(dias.map(d => { const qs = new URLSearchParams({ dia: d }); if (filial && filial !== 'TODAS') qs.set('filial', filial); return fetch('/api/brasileirao-lances?' + qs.toString(), { cache: 'no-store' }).then(r => r.ok ? r.json() : null).catch(() => null); }));
  resp.forEach((r, i) => { if (r && Array.isArray(r.lances)) r.lances.forEach(l => { l.dia = dias[i]; lances.push(l); }); });
  lances = lances.filter(l => lanceDoCanal(l, setAS));
  if (CANAL === 'AS') lances.sort((a, b) => String(b.dia + ' ' + (b.hora || '')).localeCompare(String(a.dia + ' ' + (a.hora || ''))));
  const porNivel = {}; lances.forEach(l => { porNivel[l.nivel] = (porNivel[l.nivel] || 0) + 1; });
  return { lances, porNivel, total: lances.length, periodo: per, dias };
}

function setCanal(c) {
  CANAL = c === 'AS' ? 'AS' : 'VAREJO';
  try { localStorage.setItem('brasileirao_canal', CANAL); } catch (e) {}
  aplicaCanalNaTela();
}
function aplicaCanalNaTela() {
  const as = CANAL === 'AS';
  const cv = document.getElementById('canal-v'), ca = document.getElementById('canal-a');
  if (cv && ca) { cv.classList.toggle('on', !as); ca.classList.toggle('on', as); }
  const nomes = { lances: as ? '📋 Lances da Semana' : '📋 Lances do Dia', gabarito: as ? '🎯 Gabarito da Semana' : '🎯 Gabarito do Dia' };
  document.querySelectorAll('.nav-tabs .tab-btn').forEach(b => { const m = (b.getAttribute('onclick') || '').match(/switchTab\('(\w+)'\)/); if (m && nomes[m[1]]) b.textContent = nomes[m[1]]; });
  const set = (id, t) => { const e = document.getElementById(id); if (e) e.textContent = t; };
  set('h-vend', as ? 'Artilharia dos Vendedores AS — Autosserviço' : 'Artilharia & Desempenho dos Vendedores — Varejo');
  set('s-vend', as ? 'Jogo SEMANAL • Pontos = lances da semana + faseamento da meta (20/40/60/110% = 10/20/30/40 pts) + bônus (100% até o dia 15 = +50; até o dia 25 = +25)' : 'Você contra o seu próprio dia • Vitória no Dia (3 pts) • Bônus de Constância');
  set('h-sup', as ? 'Liga dos Supervisores AS' : 'Liga dos Supervisores Comerciais — Varejo');
  set('s-sup', as ? 'Pontos da EQUIPE no faseamento da meta do mês + bônus; lances da equipe aparecem como informação' : 'Pontos do time; o Plus de Liderança (Compromisso até 10h +5, RET +5) é o critério de desempate');
  set('h-lances', as ? 'Lances da Semana — Auditoria Completa (AS)' : 'Lances do Dia — Auditoria Completa (Varejo)');
  set('h-gab', as ? 'Gabarito da Semana — regras x lances reais (AS)' : 'Gabarito do Dia — regras x lances reais (Varejo)');
  const ra = document.getElementById('reg-as'); if (ra) ra.style.display = as ? '' : 'none';
  document.querySelectorAll('.so-varejo').forEach(e => { e.style.display = as ? 'none' : ''; });
  LANCES_CARREGADOS_HOJE = false;
  filtrarSupervisores(); filtrarVendedores(); montaRegulamentoAS();
  const ativa = (document.querySelector('.tab-content.active') || {}).id;
  if (ativa === 'tab-lances') carregarLances();
  if (ativa === 'tab-gabarito') carregarGabarito();
}

function renderVendedoresAS() {

  const A = DADOS && DADOS.as;
  if (!A || !A.vendedores) { tb.innerHTML = `<tr><td colspan="9" style="text-align:center;padding:30px;color:var(--tx-mut)">${A && A.erro ? 'A Liga AS ainda não carregou (' + esc(A.erro) + ')' : 'Sem dados da Liga AS ainda.'}</td></tr>`; return; }
  const q = (document.getElementById('busca-vend').value || '').toLowerCase(), f = document.getElementById('seletor-filial-vend').value, isBrasil = f === 'BRASIL';
  let lista = A.vendedores.filter(v => isBrasil || v.filial === f);
  if (q) lista = lista.filter(v => v.nome.toLowerCase().includes(q) || String(v.rca).includes(q) || (v.supervisor || '').toLowerCase().includes(q));
  tb.innerHTML = lista.slice(0, 150).map(v => {

    const sem = (v.semanas || []).map(s => `S${s.n}: ${s.pts_lances}`).join(' · ');
